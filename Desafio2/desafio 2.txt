/*Plantilla de funcion encargada de redimensionar un arreglo en caso de no tener espacio para mas elementos, se usa para los arreglos de 
int y string de las clases E_S y Surtidor respectivamente.*/
template <typename T>
T* redimA(T* Arreglo, int tamarreglo, int tamarreglonuevo) {
    
    T* arreglonuevo = new T[tamarreglonuevo];

    for (int i = 0; i < tamarreglo; i++) {
        arreglonuevo[i] = Arreglo[i]; 
    }

    for (int i = tamarreglo; i < tamarreglonuevo; i++) {
        arreglonuevo[i] = T(); 
    }
    
    delete[] Arreglo;

    return arreglonuevo; 
}

/*Funcion que basandose en el tipo de combustible solicitado devuelve el precio que se encuentra en un arreglo con los respectivos valores 
de cada uno.
recibe un string con el que asigna un valor al precio y lo retorna.*/
int tipoComb(string tcomb,int precios[3][3],unsigned short int region){
    int precio=0;
    if (tcomb=="regular"){
        precio=*(*(precios+region));
    }
    else if (tcomb=="premium"){
        precio=*(*(precios+region)+1);
    }
    else if (tcomb=="ecoextra"){
        precio=*(*(precios+region)+2);
    }
    return precio;
}

//Funcion que convierte una cadena de caracteres a minuscula, usado para facilitar validaciones.
string pasaramin(string cadena){
    string cadenamin="";
    for (int i=0; i<cadena.length(); i++){
        cadena[i]=tolower(cadena[i]);
        cadenamin= cadenamin+cadena[i];
    }
    return cadenamin;
}

/*Funcion que puede usarse para eliminar una linea de un archivo especifico copiando los elementos del original 
exceptuando el que se desea eliminar y luego de eso se reemplaza la copia en el original y se elimina el original.*/
void borrardearchivo(const string& nombreArchivo, const string& lineaABorrar) {
    ifstream Original;
    Original.open(nombreArchivo, ios::in);

    ofstream Copia;
    Copia.open("copia.txt", ios::out);
    if (!Original.is_open() or !Copia.is_open()) {
        cerr << "Error al abrir el archivo." << endl; 
        return; 
    }

    string copiador;
    while (getline(Original, copiador)) {
        if (copiador.find(lineaABorrar)== string::npos) { 
            Copia << copiador << endl;
        }
    }

    Original.close();
    Copia.close();

    remove(nombreArchivo.c_str());
    rename("copia.txt", nombreArchivo.c_str());
}

/*Funcion que busca en el archivo correspondiente una cadena de texto solicitada y retorna la posicion en que se encuentra o "" si no se 
encuentra*/
string buscarenarch(const string& nombreArchivo, const string& textoABuscar) {
    ifstream archivo(nombreArchivo); 

    if (!archivo.is_open()) { 
        cerr << "Error al abrir el archivo: " << nombreArchivo << endl;
        return "";
    }

    string linea;
    size_t numeroLinea = 0; 

    while (getline(archivo, linea)) {
        numeroLinea++; 
        if (linea.find(textoABuscar) != string::npos) { 
            archivo.close();
            return linea;
        }
    }
    archivo.close(); 
    return "";
}
